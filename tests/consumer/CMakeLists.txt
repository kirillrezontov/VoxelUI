cmake_minimum_required(VERSION 3.25)
project(VoxelUIConsumer LANGUAGES CXX)
find_package(VoxelUI 0.2.1 EXACT CONFIG REQUIRED COMPONENTS voxelui software)
foreach(target IN ITEMS VoxelUI::voxelui VoxelUI::software)
  if(NOT TARGET ${target})
    message(FATAL_ERROR "Installed package is missing ${target}")
  endif()
  get_target_property(imported ${target} IMPORTED)
  if(NOT imported)
    message(FATAL_ERROR "Consumer must test installed imported targets: ${target}")
  endif()
  if(VOXELUI_EXPECT_IMPORTED_CONFIG)
    string(TOUPPER "${VOXELUI_EXPECT_IMPORTED_CONFIG}" config)
    get_target_property(location ${target} IMPORTED_LOCATION_${config})
    if(NOT location OR NOT EXISTS "${location}")
      message(FATAL_ERROR "${target} has no installed IMPORTED_LOCATION_${config}: '${location}'")
    endif()
  endif()
endforeach()
add_executable(external-widget main.cpp)
target_link_libraries(external-widget PRIVATE VoxelUI::voxelui VoxelUI::software)
enable_testing()
add_test(NAME external-widget COMMAND external-widget)
