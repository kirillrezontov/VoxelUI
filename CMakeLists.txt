cmake_minimum_required(VERSION 3.25)
project(VoxelUI VERSION 0.1.0 LANGUAGES CXX)
option(VOXELUI_PREVIEW_SDL "Build interactive SDL3 preview" ON)
option(VOXELUI_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(VOXELUI_TESTS "Build tests and benchmarks" ON)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(LibXml2 2.9 REQUIRED)
find_package(Freetype 2.10 REQUIRED)
find_package(PNG 1.6 REQUIRED)
add_library(voxelui
  src/xml/document.cpp src/css/style.cpp src/layout/layout.cpp
  src/svg/svg.cpp src/text/text.cpp src/runtime/resources.cpp
  src/runtime/binding.cpp src/binding/reconcile.cpp src/binding/incremental.cpp src/events/events.cpp
  src/rendering/draw_list.cpp src/runtime/runtime.cpp)
add_library(VoxelUI::voxelui ALIAS voxelui)
target_include_directories(voxelui PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(voxelui PRIVATE LibXml2::LibXml2 Freetype::Freetype PNG::PNG)
if(MSVC)
  target_compile_options(voxelui PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(voxelui PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(VOXELUI_SANITIZERS AND NOT MSVC)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
  target_compile_options(voxelui PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(voxelui PUBLIC -fsanitize=address,undefined)
endif()
add_library(voxelui_software backends/software/software.cpp)
add_library(VoxelUI::software ALIAS voxelui_software)
target_link_libraries(voxelui_software PUBLIC voxelui PRIVATE PNG::PNG)
add_executable(voxelui-preview apps/preview/main.cpp)
target_link_libraries(voxelui-preview PRIVATE voxelui_software)
if(VOXELUI_PREVIEW_SDL)
  find_package(SDL3 QUIET CONFIG)
  if(SDL3_FOUND)
    target_link_libraries(voxelui-preview PRIVATE SDL3::SDL3)
    target_compile_definitions(voxelui-preview PRIVATE VOXELUI_SDL=1)
  else()
    message(STATUS "SDL3 unavailable: preview provides headless PNG output")
  endif()
endif()
add_executable(voxelui-validate apps/validator/main.cpp)
target_link_libraries(voxelui-validate PRIVATE voxelui)
add_executable(voxelui-host examples/mock_host.cpp)
target_link_libraries(voxelui-host PRIVATE voxelui_software)
if(VOXELUI_TESTS)
  enable_testing()
  add_executable(voxelui-tests tests/tests.cpp)
  target_link_libraries(voxelui-tests PRIVATE voxelui_software)
  target_compile_definitions(voxelui-tests PRIVATE VOXELUI_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME voxelui-tests COMMAND voxelui-tests)
  add_executable(voxelui-bench benchmarks/bench.cpp)
  target_link_libraries(voxelui-bench PRIVATE voxelui)
  target_compile_definitions(voxelui-bench PRIVATE VOXELUI_BENCH_STATS=1)
endif()
include(GNUInstallDirs)
install(TARGETS voxelui voxelui_software EXPORT VoxelUITargets)
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT VoxelUITargets NAMESPACE VoxelUI:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/VoxelUI)
set_target_properties(voxelui_software PROPERTIES EXPORT_NAME software)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/VoxelUIConfig.cmake.in
  ${CMAKE_CURRENT_BINARY_DIR}/VoxelUIConfig.cmake
  INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/VoxelUI)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/VoxelUIConfigVersion.cmake
  VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/VoxelUIConfig.cmake
  ${CMAKE_CURRENT_BINARY_DIR}/VoxelUIConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/VoxelUI)
target_compile_features(voxelui PUBLIC cxx_std_23)
target_compile_options(voxelui PUBLIC $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
